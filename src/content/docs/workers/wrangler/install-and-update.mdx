import org.telegram.telegrambots.bots.TelegramLongPollingBot;
import org.telegram.telegrambots.meta.api.methods.send.SendMessage;
import org.telegram.telegrambots.meta.api.objects.Message;
import org.telegram.telegrambots.meta.api.objects.Update;
import org.telegram.telegrambots.meta.exceptions.TelegramApiException;

import java.util.HashMap;
import java.util.Map;

public class AnonymousChatBot extends TelegramLongPollingBot {
    
    private static final Long ADMIN_ID = mk_kolivand; // جایگزین با آیدی عددی ادمین
    private Map<Long, Long> activeChats = new HashMap<>(); // نگاشت کاربران به چت‌های فعال
    
    @Override
    public void onUpdateReceived(Update update) {
        if (update.hasMessage()) {
            Message message = update.getMessage();
            Long userId = message.getFrom().getId();
            String text = message.getText();
            
            // دستورات مدیریتی برای ادمین
            if (userId.equals(ADMIN_ID) && text.startsWith("/admin")) {
                handleAdminCommand(message);
                return;
            }
            
            // شروع چت ناشناس
            if ("/start".equals(text)) {
                sendMessage(userId, "به ربات چت ناشناس خوش آمدید!\nبرای شروع چت از /newchat استفاده کنید.");
                return;
            }
            
            // شروع چت جدید
            if ("/newchat".equals(text)) {
                startNewChat(userId);
                return;
            }
            
            // پایان چت
            if ("/endchat".equals(text)) {
                endChat(userId);
                return;
            }
            
            // ارسال پیام به طرف مقابل چت
            if (activeChats.containsKey(userId)) {
                forwardMessage(userId, text);
            } else {
                sendMessage(userId, "شما در حال حاضر در چت فعالی نیستید. برای شروع از /newchat استفاده کنید.");
            }
        }
    }
    
    private void handleAdminCommand(Message message) {
        Long adminId = message.getFrom().getId();
        String[] parts = message.getText().split(" ");
        
        if (parts.length < 2) {
            sendMessage(adminId, "دستورات ادمین:\n/admin chats - مشاهده چت‌های فعال\n/admin view <chatId> - مشاهده یک چت");
            return;
        }
        
        switch (parts[1]) {
            case "chats":
                listActiveChats(adminId);
                break;
            case "view":
                if (parts.length >= 3) {
                    viewChat(adminId, Long.parseLong(parts[2]));
                }
                break;
            default:
                sendMessage(adminId, "دستور نامعتبر");
        }
    }
    
    private void listActiveChats(Long adminId) {
        StringBuilder sb = new StringBuilder("چت‌های فعال:\n");
        activeChats.forEach((user1, user2) -> {
            sb.append("چت بین ").append(user1).append(" و ").append(user2).append("\n");
        });
        sendMessage(adminId, sb.toString());
    }
    
    private void viewChat(Long adminId, Long chatId) {
        // اینجا می‌توانید تاریخچه چت را از دیتابیس یا حافظه بازیابی کنید
        sendMessage(adminId, "نمایش محتوای چت " + chatId);
    }
    
    private void startNewChat(Long userId) {
        if (activeChats.containsKey(userId)) {
            sendMessage(userId, "شما در حال حاضر در یک چت هستید. برای پایان از /endchat استفاده کنید.");
            return;
        }
        
        // یافتن یک کاربر دیگر که منتظر چت است
        Long partner = findWaitingPartner(userId);
        
        if (partner != null) {
            activeChats.put(userId, partner);
            activeChats.put(partner, userId);
            sendMessage(userId, "چت ناشناس شروع شد! هر پیامی ارسال کنید به طرف مقابل می‌رسد.");
            sendMessage(partner, "چت ناشناس شروع شد! هر پیامی ارسال کنید به طرف مقابل می‌رسد.");
        } else {
            sendMessage(userId, "در حال یافتن یک شریک چت برای شما... لطفا صبر کنید.");
            // کاربر را به لیست انتظار اضافه کنید
        }
    }
    
    private Long findWaitingPartner(Long userId) {
        // اینجا باید منطق یافتن شریک چت پیاده‌سازی شود
        // برای سادگی، یک شریک تصادفی برمی‌گردانیم
        for (Map.Entry<Long, Long> entry : activeChats.entrySet()) {
            if (entry.getValue().equals(userId)) {
                return entry.getKey();
            }
        }
        return null;
    }
    
    private void endChat(Long userId) {
        if (!activeChats.containsKey(userId)) {
            sendMessage(userId, "شما در حال حاضر در چت فعالی نیستید.");
            return;
        }
        
        Long partner = activeChats.get(userId);
        activeChats.remove(userId);
        activeChats.remove(partner);
        
        sendMessage(userId, "چت با موفقیت پایان یافت. برای شروع چت جدید از /newchat استفاده کنید.");
        sendMessage(partner, "طرف مقابل چت را پایان داد. برای شروع چت جدید از /newchat استفاده کنید.");
    }
    
    private void forwardMessage(Long senderId, String text) {
        Long receiverId = activeChats.get(senderId);
        sendMessage(receiverId, text);
        
        // ذخیره پیام برای دسترسی ادمین
        saveMessageForAdmin(senderId, receiverId, text);
    }
    
    private void saveMessageForAdmin(Long sender, Long receiver, String text) {
        // اینجا می‌توانید پیام‌ها را در دیتابیس ذخیره کنید تا ادمین بتواند آنها را ببیند
    }
    
    private void sendMessage(Long chatId, String text) {
        SendMessage message = new SendMessage();
        message.setChatId(chatId.toString());
        message.setText(text);
        
        try {
            execute(message);
        } catch (TelegramApiException e) {
            e.printStackTrace();
        }
    }
    
    @Override
    public String getBotUsername() {
        return "YourBotUsername";
    }
    
    @Override
    public String getBotToken() {
        return "YOUR_BOT_TOKEN";
    }
}
